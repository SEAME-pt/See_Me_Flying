cmake_minimum_required(VERSION 3.16)
project(cv_lab CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenCV 4 REQUIRED)
message(STATUS "OpenCV ${OpenCV_VERSION}")

# ONNX Runtime (C++), unpacked in third_party/onnxruntime -> used by programs named yolo_*
set(ORT_DIR ${CMAKE_SOURCE_DIR}/third_party/onnxruntime)
set(HAVE_ORT FALSE)
if(EXISTS ${ORT_DIR}/include/onnxruntime_cxx_api.h)
  set(HAVE_ORT TRUE)
  message(STATUS "ONNX Runtime: ${ORT_DIR}")
endif()

# Every .cpp in src/ becomes its own program: src/foo.cpp -> build/foo
file(GLOB PROGRAMS ${CMAKE_SOURCE_DIR}/src/*.cpp)
foreach(src ${PROGRAMS})
  get_filename_component(name ${src} NAME_WE)
  add_executable(${name} ${src})
  target_link_libraries(${name} ${OpenCV_LIBS})
  target_compile_options(${name} PRIVATE -Wall -Wextra)
  if(name MATCHES "^yolo_")
    if(NOT HAVE_ORT)
      message(WARNING "Skipping ${name}: ONNX Runtime not found in ${ORT_DIR}")
      set_target_properties(${name} PROPERTIES EXCLUDE_FROM_ALL TRUE)
      continue()
    endif()
    target_include_directories(${name} PRIVATE ${ORT_DIR}/include)
    target_link_libraries(${name} ${ORT_DIR}/lib/libonnxruntime.so)
    set_target_properties(${name} PROPERTIES BUILD_RPATH ${ORT_DIR}/lib)
  endif()
endforeach()
